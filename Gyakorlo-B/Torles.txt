public IActionResult TorlesTermek(int id)
{
    using (var context = new RaktarkezeloContext())
    {
        try
        {
            if (context.Termekeks.Any(t => t.Id == id))
            {
                context.Remove(new Termekek()
                {
                    Id = id
                });
                context.SaveChanges();
                return Ok("Sikeres Törlés");
            }
            else
            {
                return NotFound("Nincs ilyen azonosítójú termék termék");
            }
        }
        catch (Exception ex)
        {

            return StatusCode(400, ex.Message);     
        }
    }
}
